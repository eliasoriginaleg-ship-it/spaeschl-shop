<script>
/* PRINTFUL PREISE */
const printful = {
    '12" Schallplatte': 53.00,
    '10" Schallplatte': 51.00,
    '7" Schallplatte': 41.00,
    '5" Schallplatte': 26.00,

    'T‑Shirt': 25.99
};

/* Ziffernblock */
function addNum(n) {
    const f = document.getElementById("menge");
    f.value += n;
}
function clearNum() {
    document.getElementById("menge").value = "";
}
function backspace() {
    const f = document.getElementById("menge");
    f.value = f.value.slice(0, -1);
}

/* ENTER bestätigt */
document.addEventListener("keydown", function(e) {
    if (e.key === "Enter") {
        positionHinzufuegen();
    }
});

/* Kassenlogik */
let positionen = [];

function euro(b) {
    return b.toFixed(2).replace(".", ",") + " €";
}

function datumJetzt() {
    const d = new Date();
    const pad = n => n.toString().padStart(2,"0");
    return `${pad(d.getDate())}.${pad(d.getMonth()+1)}.${d.getFullYear()} ${pad(d.getHours())}:${pad(d.getMinutes())}:${pad(d.getSeconds())}`;
}

function positionHinzufuegen() {
    const sel = document.getElementById("artikel").value;
    const menge = parseInt(document.getElementById("menge").value, 10);

    if (!sel || isNaN(menge) || menge <= 0) return;

    const [name, preisStr] = sel.split("|");
    const preis = parseFloat(preisStr);
    const total = preis * menge;

    /* Printful-Preis */
    const einkauf = printful[name] * menge;

    /* PayPal-Gebühr */
    const gebuehr = (preis * menge * 0.0249) + (0.35 * menge);

    /* Gewinn */
    const gewinn = total - einkauf - gebuehr;

    /* Aufteilung */
    const projekt = gewinn * 0.33;
    const spende  = gewinn * 0.33;
    const privat  = gewinn * 0.34;

    positionen.push({
        menge,
        name,
        preis,
        total,
        einkauf,
        gebuehr,
        projekt,
        spende,
        privat
    });

    aktualisiereAnzeige();
}

function allesLoeschen() {
    positionen = [];
    aktualisiereAnzeige();
}

function letztePositionLoeschen() {
    if (positionen.length > 0) {
        positionen.pop();
        aktualisiereAnzeige();
    }
}

function aktualisiereAnzeige() {
    const belegEl = document.getElementById("beleg");
    const gesamtEl = document.getElementById("gesamt");

  let gesamt = 0;
    let kassa = 0;
    let gebuehren = 0;
    let projekt = 0;
    let spende = 0;
    let privat = 0;

    let lines = [];

    lines.push("========================================================");
    lines.push("                     RECHNUNGSBELEG                     ");
    lines.push("========================================================");
    lines.push("Späschl");
    lines.push("Elias Gstrein");
    lines.push("E-Mail: eliasoriginal.eg@gmail.com");
    lines.push("");
    lines.push("Datum: " + datumJetzt());
    lines.push("");
    lines.push("Nr | Menge | ArtNr | Bezeichnung               | Preis   | Total");
    lines.push("------------------------------------------------------------");

    if (positionen.length === 0) {
        lines.push("(keine Positionen)");
    } else {
        positionen.forEach((p, i) => {
            gesamt += p.total;
            kassa += p.einkauf;
            gebuehren += p.gebuehr;
            projekt += p.projekt;
            spende += p.spende;
            privat += p.privat;

            lines.push(
                `${String(i+1).padStart(2)} | ${String(p.menge).padStart(5)} |  -   | ${p.name.padEnd(25)} | ${p.preis.toFixed(2).padStart(6)} | ${p.total.toFixed(2).padStart(6)}`
            );
        });
    }

    lines.push("------------------------------------------------------------");
    lines.push(`GESAMTBETRAG.............................. ${euro(gesamt)}`);

    belegEl.textContent = lines.join("\n");
    gesamtEl.textContent = euro(gesamt);

    /* LINKS ANZEIGEN */
    document.getElementById("k_kassa").textContent = euro(kassa);
    document.getElementById("k_gebuehr").textContent = euro(gebuehren);
    document.getElementById("k_projekt").textContent = euro(projekt);
    document.getElementById("k_spende").textContent  = euro(spende);
    document.getElementById("k_privat").textContent  = euro(privat);
}

function belegAlsDatei() {
    const text = document.getElementById("beleg").textContent;
    const blob = new Blob([text], {type: "text/plain;charset=utf-8"});
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = "rechnung.txt";
    a.click();
    URL.revokeObjectURL(url);
}

aktualisiereAnzeige();
