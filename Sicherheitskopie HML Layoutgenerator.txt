<script src="https://html2canvas.hertzen.com/dist/html2canvas.min.js"></script>
<script>
let imgData = null;

/* Bild laden */
document.getElementById("bild").addEventListener("change", e => {
    const file = e.target.files[0];
    if (!file) return;
    const r = new FileReader();
    r.onload = ev => { imgData = ev.target.result; updateImages(); };
    r.readAsDataURL(file);
});

/* Bild in alle Mockups einsetzen */
function updateImages() {
    ["img-polaroid","img-h","img-q","img-wanted"].forEach(id => {
        const el = document.getElementById(id);
        if (imgData) {
            el.style.backgroundImage = `url('${imgData}')`;
            el.innerHTML = "";
        } else {
            el.style.backgroundImage = "none";
            el.innerHTML = "<div style='padding-top:120px;text-align:center;'>Kein Bild geladen.</div>";
        }
    });
}

/* Export als PNG */
function exportPNG(id, filename) {
    const el = document.getElementById(id);

    html2canvas(el, { backgroundColor:null, scale:2 }).then(src => {
        const size = 1024;
        const dst = document.createElement("canvas");
        dst.width = dst.height = size;
        const ctx = dst.getContext("2d");

        const scale = Math.min(size/src.width, size/src.height);
        const w = src.width * scale;
        const h = src.height * scale;
        const x = (size - w) / 2;
        const y = (size - h) / 2;

        ctx.drawImage(src, 0, 0, src.width, src.height, x, y, w, h);

        const a = document.createElement("a");
        a.download = filename;
        a.href = dst.toDataURL("image/png");
        a.click();
    });
}

updateImages();
</script>
